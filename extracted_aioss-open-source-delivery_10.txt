  switch (strategy) {

      return product.features.find(f => f.isKillerFeature);

      product.targetAudience = "Enterprise";
      product.pricing = "Subscription/Seat";
      break;
    case 'Platform':
      // 서드파티 개발자에게 API 개방
      product.openAPI = true;
      product.allowThirdPartyApps = true;
