대규모 빅데이터 분석에서 전수 조사가 불가능할 때, 표본(Sample) 데이터를 통해 모집단(Population)의 모수를 추론하고 가설의 타당성을 통계학적으로 검증하는 **추론 통계(Statistical Inference)**의 기초를 학습한다.

---

## 1. 중심 극한 정리 (Central Limit Theorem, CLT)

모집단의 분포 형태(균등분포, 지수분포 등)에 관계없이, 표본의 크기 $n$이 충분히 크다면($n \ge 30$) 표본 평균 $\bar{X}$의 분포는 모평균 $\mu$와 표준오차 $\frac{\sigma}{\sqrt{n}}$을 갖는 정규분포에 수렴한다:

$$\bar{X} \sim \mathcal{N}\left(\mu, \frac{\sigma^2}{n}\right) \implies Z = \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \sim \mathcal{N}(0, 1)$$

- **p-value (유의확률)**: 귀무가설 $H_0$가 참이라는 가정하에, 관측된 검정 통계량만큼 극단적인 결과가 발생할 확률.
- **판정 규칙**: $p \text{-value} < \alpha$ (보통 $\alpha = 0.05$)이면 귀무가설 $H_0$를 기각하고 대립가설 $H_1$을 채택한다.

## 3. 인터랙티브 Z-검정 통계량 & p-value 계산기

(function() {
  var sX = document.getElementById('stat-xbar-slider');
  var sN = document.getElementById('stat-n-slider');
  var valX = document.getElementById('stat-xbar-val');
  var valN = document.getElementById('stat-n-val');

  var zTxt = document.getElementById('stat-z-txt');
  var pTxt = document.getElementById('stat-p-txt');
  var decTxt = document.getElementById('stat-dec-txt');
  var svg = document.getElementById('stat-svg');

  var mu0 = 100.0, sigma = 10.0;

  function erf(x) {
    var a1 =  0.254829592, a2 = -0.284496736, a3 =  1.421413741;
    var a4 = -1.453152027, a5 =  1.061405429, p  =  0.3275911;
    var sign = (x < 0) ? -1 : 1;
    x = Math.abs(x);
    var t = 1.0 / (1.0 + p * x);
    var y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);
    return sign * y;
  }

  function cdf(z) {
    return 0.5 * (1.0 + erf(z / Math.sqrt(2)));
  }

  function render() {
    var xbar = parseFloat(sX.value);
    var n = parseInt(sN.value, 10);
    valX.textContent = xbar.toFixed(1);
    valN.textContent = n + ' 개';

    var se = sigma / Math.sqrt(n);
    var z = (xbar - mu0) / se;
    var pVal = 2.0 * (1.0 - cdf(Math.abs(z)));

    zTxt.textContent = (z >= 0 ? '+' : '') + z.toFixed(2);
    pTxt.textContent = pVal < 0.0001 ? '< 0.0001' : pVal.toFixed(4);

    if (pVal < 0.05) {
      decTxt.textContent = 'H₀ 기각 (유의미한 차이)';
      decTxt.style.color = 'var(--destructive)';
    } else {
      decTxt.textContent = 'H₀ 채택 (차이 없음)';
      decTxt.style.color = 'var(--chart-2)';
    }

    // SVG 렌더링
    svg.innerHTML = '';
    var ox = 170, oy = 100, scaleX = 40, scaleY = 180;

    // 기각역 배경 (Z < -1.96, Z > +1.96)
    var dPath = 'M ' + (ox - 3.5 * scaleX) + ' ' + oy;
    for (var zv = -3.5; zv <= 3.5; zv += 0.1) {
      var yv = (1.0 / Math.sqrt(2 * Math.PI)) * Math.exp(-0.5 * zv * zv);
      dPath += ' L ' + (ox + zv * scaleX) + ' ' + (oy - yv * scaleY);
    }
    dPath += ' L ' + (ox + 3.5 * scaleX) + ' ' + oy + ' Z';

    var pCurve = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    pCurve.setAttribute('d', dPath);
    pCurve.setAttribute('fill', 'var(--background)');
    pCurve.setAttribute('stroke', 'var(--primary)');
    pCurve.setAttribute('stroke-width', '2');
    svg.appendChild(pCurve);

    // 임계선 ±1.96
    [-1.96, 1.96].forEach(function(crit) {
      var l = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      l.setAttribute('x1', ox + crit * scaleX); l.setAttribute('y1', 20);
      l.setAttribute('x2', ox + crit * scaleX); l.setAttribute('y2', oy);
      l.setAttribute('stroke', 'var(--destructive)');
      l.setAttribute('stroke-dasharray', '2 2');
      svg.appendChild(l);
    });

    // 현재 Z 위치 바늘
    var zClamped = Math.max(-3.4, Math.min(3.4, z));
    var curZLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    curZLine.setAttribute('x1', ox + zClamped * scaleX); curZLine.setAttribute('y1', 15);
    curZLine.setAttribute('x2', ox + zClamped * scaleX); curZLine.setAttribute('y2', oy);
    curZLine.setAttribute('stroke', 'var(--chart-1)');
    curZLine.setAttribute('stroke-width', '3');
    svg.appendChild(curZLine);
  }

  sX.addEventListener('input', render);
  sN.addEventListener('input', render);
  render();
})();
</script>
