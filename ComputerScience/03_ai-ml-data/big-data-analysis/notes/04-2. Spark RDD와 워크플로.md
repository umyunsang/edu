**RDD(Resilient Distributed Dataset)**는 스파크의 가장 근본적인 핵심 데이터 추상화로, 클러스터의 여러 노드에 걸쳐 분할되어 병렬 처리될 수 있는 **불변(Immutable) 분산 객체 컬렉션**이다.

---

## 1. RDD의 5대 핵심 속성과 계통도(Lineage) 복구 메커니즘

1. **파티션 목록(List of Partitions)**: 대규모 데이터가 쪼개져 있는 물리적 블록 조각들.
2. **부모 RDD에 대한 의존성 목록(List of Dependencies)**: RDD가 어떤 변환을 거쳐 생성되었는지 기록하는 계통도(Lineage DAG).
3. **연산 함수(Compute Function)**: 각 파티션 데이터를 생성하기 위한 함수.
4. **파티셔너(Partitioner - Key-Value RDD)**: HashPartitioner 또는 RangePartitioner.
5. **선호 위치 목록(Preferred Locations)**: 데이터 지역성(Data Locality)을 고려하여 블록이 위치한 노드에서 연산을 수행.

> **장애 복구 (Fault Tolerance)**: 특정 노드의 메모리 파티션이 손실되더라도, 데이터를 복제해 두지 않고 오직 계통도(Lineage DAG)를 거슬러 올라가 손실된 파티션 연산만 재실행(Re-computation)함으로써 0%의 복제 스토리지 오버헤드로 결함을 극복한다.

(function() {
  var svg = document.getElementById('rdd-dag-svg');

      var b = document.getElementById('rdd-btn-' + i);

    // 부모 파티션 3개 (x=50), 자식 파티션 3개 (x=270)
    var pY = [25, 65, 105];

    // 부모 RDD 박스
    for (var p = 0; p < 3; p++) {
      var rP = document.createElementNS('http://www.w3.org/2000/svg', 'rect');

      rP.setAttribute('rx', 3);
      rP.setAttribute('fill', 'var(--card)');
      rP.setAttribute('stroke', 'var(--chart-1)');
      svg.appendChild(rP);

      var tP = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      tP.setAttribute('x', 60); tP.setAttribute('y', pY[p] + 3);
      tP.setAttribute('font-size', '9'); tP.setAttribute('font-weight', '700');
      tP.setAttribute('text-anchor', 'middle'); tP.setAttribute('fill', 'var(--foreground)');
      tP.textContent = 'Part ' + (p + 1);
      svg.appendChild(tP);
    }

    // 자식 RDD 박스

      var rC = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
      rC.setAttribute('x', 250); rC.setAttribute('y', pY[c] - 12);
      rC.setAttribute('width', 60); rC.setAttribute('height', 24);

          lWide.setAttribute('stroke-width', '1');
          lWide.setAttribute('opacity', '0.6');
          svg.appendChild(lWide);
        }
      }
    }
  }

    (function(idx) {
      document.getElementById('rdd-btn-' + idx).addEventListener('click', function() { render(idx); });
    })(btnI);

  render(0);
