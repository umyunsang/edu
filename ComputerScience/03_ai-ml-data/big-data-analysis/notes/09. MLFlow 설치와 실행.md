## 2. Python 자동 로깅 (`mlflow.autolog()`) 파이프라인

```python
import mlflow
import mlflow.sklearn
from sklearn.ensemble import RandomForestClassifier

# 1. 원격 또는 로컬 추적 서버 연결
mlflow.set_tracking_uri("http://localhost:5000")
mlflow.set_experiment("Customer_Churn_Prediction")

with mlflow.start_run(run_name="RF_n_estimators_100"):
    # 하이퍼파라미터 수동 로깅
    mlflow.log_param("n_estimators", 100)
    mlflow.log_param("max_depth", 8)

    model = RandomForestClassifier(n_estimators=100, max_depth=8)
    model.fit(X_train, y_train)

    val_acc = model.score(X_val, y_val)
    # 메트릭 로깅
    mlflow.log_metric("val_accuracy", val_acc)

    # 모델 가중치 아티팩트 저장
    mlflow.sklearn.log_model(model, "random_forest_model")
```

(function() {
  var runs = [
    { loss: 0.452, acc: 84.2, stage: 'Staging' },
    { loss: 0.185, acc: 93.8, stage: 'None' },
    { loss: 0.112, acc: 95.6, stage: 'Production (Champion)' }
  ];

  var curIdx = 0;
  var txtLoss = document.getElementById('ml-txt-loss');
  var txtAcc = document.getElementById('ml-txt-acc');
  var txtStage = document.getElementById('ml-txt-stage');
  var btnPromote = document.getElementById('ml-btn-promote');

  function render(idx) {
    curIdx = idx;
    for (var i = 1; i <= 3; i++) {
      var b = document.getElementById('ml-run-' + i);
      if (i === idx + 1) {
        b.style.background = 'var(--primary)'; b.style.color = '#fff';
      } else {
        b.style.background = 'transparent'; b.style.color = 'var(--foreground)';
      }
    }
    var r = runs[idx];
    txtLoss.textContent = r.loss.toFixed(3);
    txtAcc.textContent = r.acc.toFixed(1) + ' %';
    txtStage.textContent = r.stage;
  }

  for (var k = 1; k <= 3; k++) {
    (function(idx) {
      document.getElementById('ml-run-' + idx).addEventListener('click', function() { render(idx - 1); });
    })(k);
  }

  btnPromote.addEventListener('click', function() {
    runs.forEach(function(r) { if (r.stage === 'Production (Champion)') r.stage = 'Archived'; });
    runs[curIdx].stage = 'Production (Champion)';
    render(curIdx);
  });

  render(0);
})();
</script>
