실시간 스트리밍 환경에서는 데이터의 총 크기 $N$이 무한에 수렴하므로 전체 데이터를 메모리에 적재할 수 없다. 따라서 **단 한 번의 순회(One-Pass)**와 **극소한 고정 메모리($O(1)$ 또는 $O(\log N)$)**만을 사용하는 **확률적 알고리즘(Probabilistic Data Structures)**이 핵심적이다.

## 2. 블룸 필터 (Bloom Filter)의 수학적 원리와 False Positive 확률

크기 $m$의 비트 배열과 $k$개의 독립된 해시 함수 $h_1, \dots, h_k$를 사용:

1. **원소 삽입**: 원소 $x$에 대해 모든 $h_i(x) \pmod m$ 위치의 비트를 $1$로 설정.
2. **원소 존재 질의**: 모든 $h_i(y)$ 위치의 비트가 $1$인지 검사.
   - 하나라도 $0$이면: **"확실히 존재하지 않음" (False Negative = 0)**
   - 모두 $1$이면: **"존재할 가능성 높음" (False Positive $\epsilon$ 존재)**

$$P(\text{False Positive}) \approx \left( 1 - e^{-kn/m} \right)^k$$
(최적 해시 함수 개수: $k = \frac{m}{n} \ln 2 \approx 0.693 \frac{m}{n}$)

<script>
(function() {
  var m = 16;
  var bitArray = new Array(m).fill(0);
  var inWord = document.getElementById('bf-input-word');
  var bitGrid = document.getElementById('bf-bit-grid');
  var statusTxt = document.getElementById('bf-status-txt');

  function hash1(str) { var h = 0; for (var i = 0; i < str.length; i++) h = (h * 31 + str.charCodeAt(i)) % m; return Math.abs(h); }
  function hash2(str) { var h = 5381; for (var i = 0; i < str.length; i++) h = ((h << 5) + h + str.charCodeAt(i)) % m; return Math.abs(h); }
  function hash3(str) { var h = 0; for (var i = 0; i < str.length; i++) h = (h * 17 + str.charCodeAt(i)) % m; return Math.abs(h); }

  function getHashes(str) {
    return [hash1(str), hash2(str), hash3(str)];
  }

  function renderBits(highlights) {
    bitGrid.innerHTML = '';
    for (var i = 0; i < m; i++) {
      var cell = document.createElement('div');
      cell.style.padding = '6px 0';
      cell.style.fontSize = '11px';
      cell.style.fontFamily = 'monospace';
      cell.style.fontWeight = '700';
      cell.style.borderRadius = '2px';
      cell.style.border = '1px solid var(--border)';

      var isHi = highlights && highlights.indexOf(i) !== -1;
      if (isHi) {
        cell.style.background = 'var(--primary)';
        cell.style.color = '#fff';
      } else if (bitArray[i] === 1) {
        cell.style.background = 'var(--chart-1)';
        cell.style.color = '#fff';
      } else {
        cell.style.background = 'var(--card)';
        cell.style.color = 'var(--muted-foreground)';
      }
      cell.textContent = bitArray[i];
      bitGrid.appendChild(cell);
    }
  }

  document.getElementById('bf-btn-add').addEventListener('click', function() {
    var w = inWord.value.trim().toLowerCase();
    if (!w) return;
    var h = getHashes(w);
    h.forEach(function(idx) { bitArray[idx] = 1; });
    renderBits(h);
    statusTxt.textContent = '✅ "' + w + '" 삽입 완료 ➔ 해시 비트 인덱스: [' + h.join(', ') + ']';
    statusTxt.style.color = 'var(--chart-2)';
  });

  document.getElementById('bf-btn-check').addEventListener('click', function() {
    var w = inWord.value.trim().toLowerCase();
    if (!w) return;
    var h = getHashes(w);
    renderBits(h);
    var allOne = h.every(function(idx) { return bitArray[idx] === 1; });
    if (allOne) {
      statusTxt.textContent = '🟢 "' + w + '" 발견됨 (확률적 존재, False Positive 가능성 있음) ➔ 비트: [' + h.join(', ') + ']';
      statusTxt.style.color = 'var(--primary)';
    } else {
      statusTxt.textContent = '🔴 "' + w + '" 확실히 없음 (False Negative 없음) ➔ 비트: [' + h.join(', ') + ']';
      statusTxt.style.color = 'var(--destructive)';
    }
  });

  document.getElementById('bf-btn-reset').addEventListener('click', function() {
    bitArray.fill(0);
    renderBits([]);
    statusTxt.textContent = '비트맵이 초기화되었습니다.';
    statusTxt.style.color = 'var(--muted-foreground)';
  });

  renderBits([]);
})();
</script>
