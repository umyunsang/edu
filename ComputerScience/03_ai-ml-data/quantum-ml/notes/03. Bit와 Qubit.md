고전 컴퓨터의 최소 정보 단위인 **비트(Bit)**는 $0$ 또는 $1$의 불연속적인 두 상태 중 하나에만 머무를 수 있는 스위치이지만, 양자 컴퓨터의 **큐비트(Qubit)**는 2차원 복소 벡터 공간 상의 연속적인 회전 상태를 가진다.

---

## 2. 큐비트 상태 벡터와 극좌표 표현

$$|\psi\rangle = \cos\left(\frac{\theta}{2}\right) |0\rangle + e^{i\phi} \sin\left(\frac{\theta}{2}\right) |1\rangle$$

- $\theta \in [0, \pi]$: 북극($|0\rangle$)과 남극($|1\rangle$) 사이의 위도 각도 (측정 확률 비율 결정).
- $\phi \in [0, 2\pi)$: 적도면 상의 경도 위상(Relative Phase) 각도 (양자 간섭 결정).

---

## 3. 인터랙티브 큐비트 각도($\theta, \phi$) ➔ 확률 계산기

(function() {
  var sTh = document.getElementById('bq-th-slider');
  var sPhi = document.getElementById('bq-phi-slider');
  var thVal = document.getElementById('bq-th-val');
  var phiVal = document.getElementById('bq-phi-val');

  var ketTxt = document.getElementById('bq-ket-txt');
  var p0Txt = document.getElementById('bq-p0-txt');
  var p1Txt = document.getElementById('bq-p1-txt');
  var p0Bar = document.getElementById('bq-p0-bar');
  var p1Bar = document.getElementById('bq-p1-bar');

  function render() {
    var degTh = parseFloat(sTh.value);
    var degPhi = parseFloat(sPhi.value);
    thVal.textContent = degTh.toFixed(0) + '°';
    phiVal.textContent = degPhi.toFixed(0) + '°';

    var radTh = degTh * Math.PI / 180.0;
    var radPhi = degPhi * Math.PI / 180.0;

    var a = Math.cos(radTh / 2.0);
    var b = Math.sin(radTh / 2.0);

    var p0 = a * a;
    var p1 = b * b;

    p0Txt.textContent = (p0 * 100).toFixed(1) + ' %';
    p1Txt.textContent = (p1 * 100).toFixed(1) + ' %';
    p0Bar.style.width = (p0 * 100).toFixed(1) + '%';
    p1Bar.style.width = (p1 * 100).toFixed(1) + '%';

    var bReal = b * Math.cos(radPhi);
    var bImag = b * Math.sin(radPhi);

    var strB = '';
    if (Math.abs(bImag) < 0.001) {
      strB = (bReal >= 0 ? '+ ' : '- ') + Math.abs(bReal).toFixed(3);
    } else {
      strB = '+ (' + bReal.toFixed(3) + (bImag >= 0 ? ' + ' : ' - ') + Math.abs(bImag).toFixed(3) + 'i)';
    }

    ketTxt.textContent = a.toFixed(3) + ' |0⟩ ' + strB + ' |1⟩';
  }

  sTh.addEventListener('input', render);
  sPhi.addEventListener('input', render);
  render();
})();
</script>
