(function() {
  var stages = [
    {
      title: 'Stage 1: 공학설계와 AI 시스템 관점 (Note 01)',
      desc: '공학설계의 정의, CDIO 프로세스, 코드 중심 SW와 데이터 중심 AI 시스템의 패러다임 차이, 5대 설계 성숙도 진단.',

      ]
    },
    {
      title: 'Stage 3: 공학 윤리와 지식재산권 보호 (Note 03)',
      desc: 'AI 공정성/설명가능성/책임성, 특허권(Patent) vs 저작권(Copyright) vs 영업비밀(Trade Secret) 자산 보호.',
      notes: [
        '03. 공학 윤리와 지식재산권 - AI 신뢰성과 특허·저작권·영업비밀 보호'
      ]
    },
    {
      title: 'Stage 4: 팀 프로젝트 계획 및 비즈니스 모델 (Notes 04, 05)',
      desc: 'WBS 작업 분할, 간트 차트 주차별 마일스톤, 위험 관리 매트릭스, AI 비즈니스 모델 캔버스(BMC) 및 손익분기점(BEP).',
      notes: [
        '04. 팀 설계와 설계 계획서 - WBS, 간트 차트와 위험 관리 매트릭스',
        '05. AI 아이디어와 보호 전략 - 비즈니스 모델 캔버스와 특허 선행기술조사'
      ]
    }

  var sTitle = document.getElementById('as-title');
  var sDesc = document.getElementById('as-desc');
  var sList = document.getElementById('as-list');

  function setStage(idx) {
    for (var i = 1; i <= 4; i++) {
      var btn = document.getElementById('as-btn-' + i);
      if (i === idx + 1) {
        btn.style.background = 'var(--primary)';
        btn.style.color = '#fff';
      } else {
        btn.style.background = 'transparent';
        btn.style.color = 'var(--foreground)';
      }
    }

    var st = stages[idx];
    sTitle.textContent = st.title;
    sDesc.textContent = st.desc;
    sList.innerHTML = '';

    st.notes.forEach(function(n) {
      var d = document.createElement('div');
      d.style.padding = '8px 12px';
      d.style.background = 'var(--card)';
      d.style.border = '1px solid var(--border)';

      d.style.fontWeight = '600';
      d.style.color = 'var(--foreground)';
      d.textContent = '📄 ' + n;
      sList.appendChild(d);
    });
  }

  for (var k = 1; k <= 4; k++) {
    (function(idx) {
      document.getElementById('as-btn-' + (idx + 1)).addEventListener('click', function() {
        setStage(idx);
      });
    })(k - 1);
  }

  setStage(0);
})();
</script>
