인공지능 및 딥러닝 모델의 최적화 이론을 정립하기 위해 필수적인 **행렬 대수의 핵심 성질**과 **선형 회귀의 최소자승법(Ordinary Least Squares, OLS) 정규 방정식**을 엄밀한 수학적 유도로 정리한다.

---

## 1. 행렬 곱의 전치 성질 $(AB)^T = B^T A^T$ 대수적 증명

행렬 $\mathbf{A} \in \mathbb{R}^{m \times k}$와 $\mathbf{B} \in \mathbb{R}^{k \times n}$에 대하여:

1. $\mathbf{C} = \mathbf{A}\mathbf{B}$의 $(i, j)$번째 원소는 정의에 의해 다음과 같다:
   $$C_{ij} = \sum_{r=1}^k A_{ir} B_{rj}$$
2. 전치 행렬 $\mathbf{C}^T$의 $(i, j)$번째 원소는 원래 행렬 $\mathbf{C}$의 $(j, i)$번째 원소이므로:
   $$(\mathbf{C}^T)_{ij} = C_{ji} = \sum_{r=1}^k A_{jr} B_{ri} = \sum_{r=1}^k B_{ri} A_{jr}$$
3. 한편, $B_{ri} = (\mathbf{B}^T)_{ir}$이고 $A_{jr} = (\mathbf{A}^T)_{rj}$이므로:
   $$(\mathbf{C}^T)_{ij} = \sum_{r=1}^k (\mathbf{B}^T)_{ir} (\mathbf{A}^T)_{rj} = (\mathbf{B}^T \mathbf{A}^T)_{ij}$$
4. 모든 $(i, j)$ 성분에 대해 성립하므로, **$(\mathbf{A}\mathbf{B})^T = \mathbf{B}^T \mathbf{A}^T$**가 증명된다.

---

## 2. 최소자승법(Least Squares) 정규 방정식(Normal Equation) 행렬 미분 유도

목표는 잔차 제곱합 손실 $E(\mathbf{w})$를 최소화하는 최적 가중치 벡터 $\mathbf{w}^*$를 구하는 것이다:

$$E(\mathbf{w}) = \|\mathbf{y} - \mathbf{X}\mathbf{w}\|^2 = (\mathbf{y} - \mathbf{X}\mathbf{w})^T (\mathbf{y} - \mathbf{X}\mathbf{w})$$

### 1) 손실 함수 전개

$$E(\mathbf{w}) = \mathbf{y}^T \mathbf{y} - \mathbf{y}^T \mathbf{X}\mathbf{w} - \mathbf{w}^T \mathbf{X}^T \mathbf{y} + \mathbf{w}^T \mathbf{X}^T \mathbf{X}\mathbf{w}$$
스칼라이므로 $\mathbf{y}^T \mathbf{X}\mathbf{w} = \mathbf{w}^T \mathbf{X}^T \mathbf{y}$이다:
$$E(\mathbf{w}) = \mathbf{y}^T \mathbf{y} - 2 \mathbf{w}^T \mathbf{X}^T \mathbf{y} + \mathbf{w}^T (\mathbf{X}^T \mathbf{X}) \mathbf{w}$$

### 2) 가중치 벡터 $\mathbf{w}$에 대한 행렬 미분

행렬 미분 공식 $\frac{\partial (\mathbf{a}^T \mathbf{w})}{\partial \mathbf{w}} = \mathbf{a}$, $\frac{\partial (\mathbf{w}^T \mathbf{A} \mathbf{w})}{\partial \mathbf{w}} = 2\mathbf{A}\mathbf{w}$ (A가 대칭행렬일 때)를 적용:
$$\nabla_{\mathbf{w}} E(\mathbf{w}) = -2\mathbf{X}^T \mathbf{y} + 2\mathbf{X}^T \mathbf{X}\mathbf{w} = \mathbf{0}$$

### 3) 최적해 (정규 방정식)

$$\mathbf{X}^T \mathbf{X}\mathbf{w} = \mathbf{X}^T \mathbf{y} \implies \mathbf{w}^* = (\mathbf{X}^T \mathbf{X})^{-1} \mathbf{X}^T \mathbf{y}$$

(function() {
  var sSlope = document.getElementById('ols-slope-slider');
  var sNoise = document.getElementById('ols-noise-slider');
  var slopeVal = document.getElementById('ols-slope-val');
  var noiseVal = document.getElementById('ols-noise-val');

  var wTxt = document.getElementById('ols-w-txt');
  var bTxt = document.getElementById('ols-b-txt');
  var sseTxt = document.getElementById('ols-sse-txt');
  var svg = document.getElementById('ols-svg');

  var fixedX = [-2.0, -1.5, -1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0];
  var fixedNoise = [-0.3, 0.5, -0.2, 0.4, -0.1, -0.4, 0.3, -0.5, 0.2];

  function render() {
    var trueM = parseFloat(sSlope.value);
    var noise = parseFloat(sNoise.value);
    slopeVal.textContent = trueM.toFixed(2);
    noiseVal.textContent = noise.toFixed(2);

    var N = fixedX.length;
    var sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
    var pts = [];

    for (var i = 0; i < N; i++) {
      var x = fixedX[i];
      var y = trueM * x + noise * fixedNoise[i];
      pts.push({ x: x, y: y });
      sumX += x; sumY += y;
      sumXY += x * y; sumX2 += x * x;
    }

    var wOpt = (N * sumXY - sumX * sumY) / (N * sumX2 - sumX * sumX);
    var bOpt = (sumY - wOpt * sumX) / N;

    var sse = 0;
    for (var j = 0; j < N; j++) {
      var yHat = wOpt * pts[j].x + bOpt;
      sse += Math.pow(pts[j].y - yHat, 2);
    }

    wTxt.textContent = wOpt.toFixed(3);
    bTxt.textContent = bOpt.toFixed(3);
    sseTxt.textContent = sse.toFixed(3);

    // SVG 렌더링
    svg.innerHTML = '';
    var ox = 170, oy = 80, scaleX = 35, scaleY = 16;

    // 축
    var axX = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    axX.setAttribute('x1', 20); axX.setAttribute('y1', oy);
    axX.setAttribute('x2', 320); axX.setAttribute('y2', oy);
    axX.setAttribute('stroke', 'var(--border)');
    svg.appendChild(axX);

    var axY = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    axY.setAttribute('x1', ox); axY.setAttribute('y1', 10);
    axY.setAttribute('x2', ox); axY.setAttribute('y2', 150);
    axY.setAttribute('stroke', 'var(--border)');
    svg.appendChild(axY);

    // 회귀선 (x = -2.5 ~ +2.5)
    var x1 = -2.5, y1 = wOpt * x1 + bOpt;
    var x2 = 2.5, y2 = wOpt * x2 + bOpt;
    var rLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    rLine.setAttribute('x1', ox + x1 * scaleX); rLine.setAttribute('y1', oy - y1 * scaleY);
    rLine.setAttribute('x2', ox + x2 * scaleX); rLine.setAttribute('y2', oy - y2 * scaleY);
    rLine.setAttribute('stroke', 'var(--primary)'); rLine.setAttribute('stroke-width', '2.5');
    svg.appendChild(rLine);

    // 점 및 잔차 선분
    for (var k = 0; k < N; k++) {
      var px = ox + pts[k].x * scaleX;
      var py = oy - pts[k].y * scaleY;
      var pyHat = oy - (wOpt * pts[k].x + bOpt) * scaleY;

      var resLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      resLine.setAttribute('x1', px); resLine.setAttribute('y1', py);
      resLine.setAttribute('x2', px); resLine.setAttribute('y2', pyHat);
      resLine.setAttribute('stroke', 'var(--destructive)');
      resLine.setAttribute('stroke-dasharray', '2 2');
      svg.appendChild(resLine);

      var dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
      dot.setAttribute('cx', px); dot.setAttribute('cy', py);
      dot.setAttribute('r', 4);
      dot.setAttribute('fill', 'var(--chart-2)');
      dot.setAttribute('stroke', 'var(--background)');
      dot.setAttribute('stroke-width', '1');
      svg.appendChild(dot);
    }
  }

  sSlope.addEventListener('input', render);
  sNoise.addEventListener('input', render);
  render();
})();
</script>
