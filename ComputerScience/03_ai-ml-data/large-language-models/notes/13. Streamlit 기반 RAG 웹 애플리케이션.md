**Streamlit**은 복잡한 프론트엔드(HTML/CSS/JS) 개발 없이 순수 파이썬 코드만으로 직관적인 대시보드와 대화형 RAG 챗봇 서비스를 신속하게 구축할 수 있는 오픈소스 프레임워크이다.

---

## 1. Streamlit 챗봇 핵심 컴포넌트

| 컴포넌트 | 역할 및 사용법 |
| :--- | :--- |
| `st.session_state` | 사용자의 페이지 리로드 간 대화 내역(`messages`) 및 벡터 인덱스 메모리 유지 |
| `st.chat_message("user"/"assistant")` | 말풍선 형태의 대화 블록 렌더링 |
| `st.chat_input("질문을 입력하세요")` | 하단 고정 채팅 입력 인터페이스 |
| `st.spinner("답변 생성 중...")` | LLM 추론 및 검색 중 로딩 인디케이터 표시 |

## 2. Streamlit + LangChain RAG 통합 소스 코드 (`app.py`)

```python
import streamlit as st
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain.chains import create_retrieval_chain
from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain_core.prompts import ChatPromptTemplate
from dotenv import load_dotenv

load_dotenv()

st.set_page_config(page_title="RAG 지능형 Q&A 서비스", page_icon="🤖")
st.title("🤖 사내 문서 기반 지능형 질의응답 시스템")

# 1. 세션 상태 초기화
if "messages" not in st.session_state:
    st.session_state.messages = []

# 2. RAG 체인 캐싱 (반복 로딩 방지)
@st.cache_resource
def get_rag_chain():
    embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    vectorstore = Chroma(persist_directory="./chroma_db", embedding_function=embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 3})
    
    system_prompt = (
        "당신은 사내 지식 기반 질의응답 AI입니다. "
        "다음 검색된 문서(context)를 바탕으로 정확하게 답변하세요:\n\n{context}"
    )
    prompt = ChatPromptTemplate.from_messages([
        ("system", system_prompt),
        ("human", "{input}")
    ])
    
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.2)
    question_answer_chain = create_stuff_documents_chain(llm, prompt)
    return create_retrieval_chain(retriever, question_answer_chain)

rag_chain = get_rag_chain()

# 3. 이전 대화 기록 렌더링
for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"])

# 4. 사용자 입력 처리
if user_query := st.chat_input("질문을 입력하세요..."):
    st.session_state.messages.append({"role": "user", "content": user_query})
    with st.chat_message("user"):
        st.markdown(user_query)

    with st.chat_message("assistant"):
        with st.spinner("지식 베이스 검색 및 답변 생성 중..."):
            response = rag_chain.invoke({"input": user_query})
            answer = response["answer"]
            st.markdown(answer)
            
            # 근거 문서 아코디언 표시
            with st.expander("📚 참조한 근거 문서 확인"):
                for i, doc in enumerate(response["context"]):
                    st.caption(f"**[문서 {i+1}]** {doc.page_content[:200]}...")

    st.session_state.messages.append({"role": "assistant", "content": answer})
```

```html preview
    </div>
  </div>

  <!-- 입력창 -->
  <div style="display:flex;gap:6px;margin-top:10px">
    <input id="st-inp-text" type="text" placeholder="예: 재택근무 신청 절차 알려줘" style="flex:1;padding:6px 10px;background:var(--background);border:1px solid var(--border);color:var(--foreground);border-radius:var(--radius);font-size:11px" />
    <button id="st-send-btn" style="padding:6px 14px;background:var(--primary);color:#fff;border:none;border-radius:var(--radius);font-size:11px;font-weight:700;cursor:pointer">전송</button>
  </div>

<script>
(function() {
  var box = document.getElementById('st-msg-box');
  var inp = document.getElementById('st-inp-text');
  var btn = document.getElementById('st-send-btn');

  function send() {
    var q = inp.value.trim();
    if (!q) return;

    // Add user message
    var uDiv = document.createElement('div');
    uDiv.style.padding = '6px 10px';
    uDiv.style.background = 'var(--primary)';
    uDiv.style.color = '#fff';
    uDiv.style.borderRadius = '6px';
    uDiv.style.fontSize = '11px';
    uDiv.style.alignSelf = 'flex-end';
    uDiv.textContent = q;
    box.appendChild(uDiv);

    inp.value = '';

    // Add assistant response
    setTimeout(function() {
      var aDiv = document.createElement('div');
      aDiv.style.padding = '6px 10px';
      aDiv.style.background = 'var(--card)';
      aDiv.style.border = '1px solid var(--chart-2)';
      aDiv.style.borderRadius = '6px';
      aDiv.style.fontSize = '11px';
      aDiv.style.alignSelf = 'flex-start';
      aDiv.innerHTML = '<strong>AI 답변 (RAG 근거):</strong> [사내 복무규정 제7조] 재택근무는 주 2회까지 그룹웨어 결재를 통해 신청 가능합니다.';
      box.appendChild(aDiv);
      box.scrollTop = box.scrollHeight;
    }, 400);

    box.scrollTop = box.scrollHeight;
  }

  btn.addEventListener('click', send);
  inp.addEventListener('keydown', function(e) { if (e.key === 'Enter') send(); });
})();
</script>
```
