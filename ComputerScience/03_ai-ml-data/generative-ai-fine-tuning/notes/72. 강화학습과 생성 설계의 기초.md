대규모 생성 모델의 가치 정렬(Alignment)은 복잡한 **PPO 기반 RLHF(보상 모델 + 액터 + 크리틱 + 참조 모델 4개 신경망 동시 구동)**에서 출발하여, 수학적으로 동등하면서도 훈련이 극도로 안정적인 **DPO(Direct Preference Optimization)**로 진화하였다.

## 2. DPO 손실 함수의 수학적 유도

보상 $r(x, y) = \beta \log \frac{\pi_\theta(y|x)}{\pi_{\text{ref}}(y|x)}$ 관계를 Bradley-Terry 선호 모델 $P(y_w \succ y_l \mid x) = \sigma(r(x, y_w) - r(x, y_l))$에 대입하여 도출:

$$\mathcal{L}_{\text{DPO}}(\theta; \pi_{\text{ref}}) = -\mathbb{E}_{(x, y_w, y_l)}\left[ \log \sigma\left( \beta \log \frac{\pi_\theta(y_w|x)}{\pi_{\text{ref}}(y_w|x)} - \beta \log \frac{\pi_\theta(y_l|x)}{\pi_{\text{ref}}(y_l|x)} \right) \right]$$

- $\beta$: 기준 모델 $\pi_{\text{ref}}$로부터의 과도한 이탈을 억제하는 정규화 하이퍼파라미터 (보통 $0.1 \sim 0.5$).

## 3. 인터랙티브 DPO 암시적 보상 마진($\Delta r$) 및 손실 계산기

선호 답변($y_w$)과 비선호 답변($y_l$)에 대한 정책 확률비 로그 차이($\Delta \log \pi$)를 슬라이더로 조절하면 DPO 손실 $\mathcal{L}_{\text{DPO}}$과 선호 확률 $P(y_w \succ y_l)$이 실시간으로 계산된다.

<script>
(function() {
  var sDel = document.getElementById('dpo-del-slider');
  var delVal = document.getElementById('dpo-del-val');
  var rewTxt = document.getElementById('dpo-rew-txt');
  var probTxt = document.getElementById('dpo-prob-txt');
  var lossTxt = document.getElementById('dpo-loss-txt');

  var beta = 0.1;

  function render() {
    var delta = parseFloat(sDel.value);
    delVal.textContent = 'Δ = ' + (delta >= 0 ? '+' : '') + delta.toFixed(1);

    var impRew = beta * delta;
    // P = 1 / (1 + exp(-delta))
    var prob = 1.0 / (1.0 + Math.exp(-delta));
    // Loss = -log(sigmoid(delta)) = log(1 + exp(-delta))
    var loss = Math.log(1.0 + Math.exp(-delta));

    rewTxt.textContent = (impRew >= 0 ? '+' : '') + impRew.toFixed(2);
    probTxt.textContent = (prob * 100).toFixed(1) + ' %';
    lossTxt.textContent = loss.toFixed(3);

    probTxt.style.color = prob > 0.7 ? 'var(--chart-2)' : (prob > 0.4 ? 'var(--primary)' : 'var(--destructive)');
  }

  sDel.addEventListener('input', render);
  render();
})();
</script>
