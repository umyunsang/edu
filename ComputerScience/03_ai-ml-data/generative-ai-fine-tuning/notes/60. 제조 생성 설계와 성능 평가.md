**제조 생성 설계(Generative Design in Manufacturing)**는 디자이너가 형상을 직접 그리는 대신, 하중 조건(Load), 고정점(Boundary Condition), 목표 경량화율을 입력하면 AI가 구조적 안전성을 만족하는 최적의 바이오모픽(Biomorphic) 형상을 자율 생성하는 기술이다.

## 2. 폰 미세스 등가 응력 (Von Mises Stress) 공식

$$\sigma_{\text{vm}} = \sqrt{\frac{1}{2} \left[ (\sigma_x - \sigma_y)^2 + (\sigma_y - \sigma_z)^2 + (\sigma_z - \sigma_x)^2 + 6(\tau_{xy}^2 + \tau_{yz}^2 + \tau_{zx}^2) \right]}$$

---

## 3. 인터랙티브 구조물 경량화율 & 안전 계수(Safety Factor) 시뮬레이터

경량화 목표 비율(Volume Reduction: 10% ~ 70%)을 조절하면 구조물의 무게 절감량, 최대 폰 미세스 응력 및 안전 계수($\text{SF} = \sigma_{\text{yield}} / \sigma_{\text{max}}$)가 실시간으로 계산된다.

(function() {
  var sRed = document.getElementById('gd-red-slider');
  var redVal = document.getElementById('gd-red-val');
  var wtTxt = document.getElementById('gd-wt-txt');
  var strTxt = document.getElementById('gd-str-txt');
  var sfTxt = document.getElementById('gd-sf-txt');

  var yieldStress = 250.0; // 알루미늄 6061-T6 항복 강도 (MPa)

  function render() {
    var red = parseInt(sRed.value, 10);
    redVal.textContent = red + ' % 감축';

    var wt = 100.0 * (1.0 - red / 100.0);
    // 응력은 질량이 줄어들수록 비선형적으로 증가
    var stress = 60.0 / Math.pow(1.0 - red / 100.0, 0.9);
    var sf = yieldStress / stress;

    wtTxt.textContent = wt.toFixed(1) + ' kg (-' + red + '%)';
    strTxt.textContent = stress.toFixed(1) + ' MPa';
    sfTxt.textContent = sf.toFixed(2) + (sf >= 1.5 ? ' (안전)' : ' (위험)');

    if (sf >= 1.5) {
      sfTxt.style.color = 'var(--chart-2)';
    } else if (sf >= 1.0) {
      sfTxt.style.color = 'var(--chart-1)';
    } else {
      sfTxt.style.color = 'var(--destructive)';
    }
  }

  sRed.addEventListener('input', render);
  render();
})();
</script>
