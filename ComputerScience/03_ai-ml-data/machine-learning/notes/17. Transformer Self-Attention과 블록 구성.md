트랜스포머의 단일 레이어 블록은 \*\*멀티헤드 셀프 어텐션(MHA)\*\*과 \*\*포지션 와이즈 피드포워드 네트워크(FFN)\*\*를 2개의 잔차 연결(Residual Connection) 및 레이어 정규화(Layer Normalization)로 결합하여 깊은 신경망의 학습 안정성을 보장한다.

---

## 1. 멀티헤드 어텐션과 트랜스포머 서브레이어 수학적 공식화

### 1) 멀티헤드 어텐션 (Multi-Head Attention, MHA)

$h$개의 서로 다른 표현 부분공간(Subspaces)을 병렬 학습하기 위한 선형 투영:

$$\text{MultiHead}(Q, K, V) = \text{Concat}(\text{head}_1, \text{head}_2, \dots, \text{head}_h) W^O$$

$$\text{head}_i = \text{Attention}\big( Q W_i^Q, \; K W_i^K, \; V W_i^V \big)$$

- 파라미터 행렬 크기: $W_i^Q, W_i^K \in \mathbb{R}^{d_{\text{model}} \times d_k}$, $W_i^V \in \mathbb{R}^{d_{\text{model}} \times d_v}$, $W^O \in \mathbb{R}^{h d_v \times d_{\text{model}}}$.
- 통상 $d_k = d_v = d\_{\text{model}} / h$로 설정하여 단일 풀 어텐션과 동일한 계산 복잡도 $\mathcal{O}(n^2 \cdot d\_{\text{model}})$를 유지.

### 2) 점별 피드포워드 신경망 (Position-wise Feed-Forward Network)

$$\text{FFN}(\mathbf{x}) = \max(0, \mathbf{x} W_1 + \mathbf{b}_1) W_2 + \mathbf{b}_2$$

- 내부 은닉층 차원은 통상 $d\_{\text{ff}} = 4 \times d\_{\text{model}}$로 4배 확장 후 복원.

### 3) 레이어 정규화 (Layer Normalization)

$$\text{LN}(\mathbf{x}) = \frac{\mathbf{x} - \mu}{\sqrt{\sigma^2 + \epsilon}} \odot \boldsymbol{\gamma} + \boldsymbol{\beta}$$

$$\mu = \frac{1}{d}\sum_{i=1}^d x_i, \quad \sigma^2 = \frac{1}{d}\sum_{i=1}^d (x_i - \mu)^2$$

(function() {
  var in1 = document.getElementById('ln-x1');
  var in2 = document.getElementById('ln-x2');
  var in3 = document.getElementById('ln-x3');
  var in4 = document.getElementById('ln-x4');

  function update() {
    var vec = [parseFloat(in1.value)||0, parseFloat(in2.value)||0, parseFloat(in3.value)||0, parseFloat(in4.value)||0];
    var d = vec.length;
    var mean = vec.reduce(function(a,b){return a+b;}, 0) / d;
    var variance = vec.reduce(function(a,b){return a + Math.pow(b - mean, 2);}, 0) / d;
    var std = Math.sqrt(variance + 1e-6);

    var norm = vec.map(function(v){ return ((v - mean) / std).toFixed(2); });

    rStat.textContent = 'μ = ' + mean.toFixed(2) + ', σ² = ' + variance.toFixed(2);
    rVec.textContent = '[' + norm.join(', ') + ']';
  }

  [in1, in2, in3, in4].forEach(function(el){ el.addEventListener('input', update); });
  update();
})();
</script>
