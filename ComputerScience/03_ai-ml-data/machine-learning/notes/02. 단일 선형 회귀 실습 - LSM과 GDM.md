단일 선형 회귀 실습에서는 입력 데이터에 편향(Bias) 열을 결합한 행렬 $X$를 구성하고, 정규방정식(`np.linalg.inv`)과 경사 하강법 루프(`w -= alpha * grad_w`)를 코드로 구현하여 수렴 양상을 분석한다.

> [!IMPORTANT]
>
> - **경사 하강법 편미분 도출**:
>   $$\frac{\partial L}{\partial w} = -\frac{2}{N}\sum_{i=1}^N x_i (y_i - (wx_i + b)), \quad \frac{\partial L}{\partial b} = -\frac{2}{N}\sum_{i=1}^N (y_i - (wx_i + b))$$
> - **수렴 판정 기준**: 에포크 진행에 따라 손실 변화율 $\vert \Delta L\vert  < \epsilon$일 때 조기 종료(Early Stopping).

(function() {
  var xs = [-3, -1, 1, 3];
  var ys = [-1, -1, 3, 3];
  var N = xs.length;

  var w = 0.0;
  var b = 0.0;
  var epoch = 0;

  var selLr = document.getElementById('gdm-sel-lr');
  var btnStep = document.getElementById('gdm-btn-step');
  var btn10 = document.getElementById('gdm-btn-10');
  var btnReset = document.getElementById('gdm-btn-reset');

  var valEpoch = document.getElementById('gdm-epoch-val');
  var valW = document.getElementById('gdm-w-val');
  var valB = document.getElementById('gdm-b-val');
  var valLoss = document.getElementById('gdm-loss-val');

  function calcLoss() {
    var sse = 0;
    for (var i = 0; i < N; i++) {
      var err = ys[i] - (w * xs[i] + b);
      sse += err * err;
    }
    return sse / N;
  }

  function render() {
    valEpoch.textContent = epoch;
    valW.textContent = w.toFixed(3);
    valB.textContent = b.toFixed(3);
    valLoss.textContent = calcLoss().toFixed(3);
  }

  function step(lr) {
    var gradW = 0;
    var gradB = 0;
    for (var i = 0; i < N; i++) {
      var err = ys[i] - (w * xs[i] + b);
      gradW += -2 * xs[i] * err;
      gradB += -2 * err;
    }
    gradW /= N;
    gradB /= N;

    w -= lr * gradW;
    b -= lr * gradB;
    epoch++;
  }

  btnStep.addEventListener('click', function() {
    var lr = parseFloat(selLr.value);
    step(lr);
    render();
  });

  btn10.addEventListener('click', function() {
    var lr = parseFloat(selLr.value);
    for (var i = 0; i < 10; i++) step(lr);
    render();
  });

  btnReset.addEventListener('click', function() {
    w = 0.0; b = 0.0; epoch = 0;
    render();
  });

  render();
})();
</script>

## 2. 정규방정식 NumPy 5줄 구현

```python
import numpy as np

# 1. 데이터 정의 및 Bias 열(1) 결합
X = np.array([[-3, 1], [-1, 1], [1, 1], [3, 1]])
Y = np.array([[-1], [-1], [3], [3]])

# 2. 정규방정식 theta = (X^T X)^-1 X^T Y 계산
XT_X = np.dot(X.T, X)
XT_Y = np.dot(X.T, Y)
theta = np.dot(np.linalg.inv(XT_X), XT_Y)

print("최적 파라미터 [w, b]:", theta.ravel())  # [0.8, 1.0]
```
