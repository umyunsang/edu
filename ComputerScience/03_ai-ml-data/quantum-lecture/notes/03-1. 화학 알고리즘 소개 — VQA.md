NISQ(Noisy Intermediate-Scale Quantum) 시대에 가장 유망한 양자 화학 응용 알고리즘인 **VQE(Variational Quantum Eigensolver)**는 분자의 전자 구조 해밀토니안의 **바닥 상태 에너지(Ground State Energy $E_0$)**를 양자-고전 하이브리드 방식으로 계산한다.

---

## 1. 변분 원리 (Variational Principle)의 수학적 기초

임의의 파라미터화된 양자 상태 $|\psi(\boldsymbol{\theta})\rangle$에 대한 해밀토니안 $\hat{H}$의 기댓값 $E(\boldsymbol{\theta})$는 시스템의 실제 최저 고유값(바닥 상태 에너지 $E_0$)보다 항상 크거나 같다:

$$E(\boldsymbol{\theta}) = \frac{\langle\psi(\boldsymbol{\theta})|\hat{H}|\psi(\boldsymbol{\theta})\rangle}{\langle\psi(\boldsymbol{\theta})|\psi(\boldsymbol{\theta})\rangle} \ge E_0$$

## 2. 수소 분자 ($H_2$) 2-큐비트 해밀토니안 파울리 분해

조던-위그너(Jordan-Wigner) 변환을 통해 페르미온 해밀토니안을 파울리 텐서 합으로 변환:

$$\hat{H} = g_0 \mathbf{I} + g_1 \mathbf{Z}_0 + g_2 \mathbf{Z}_1 + g_3 \mathbf{Z}_0\mathbf{Z}_1 + g_4 \mathbf{X}_0\mathbf{X}_1 + g_5 \mathbf{Y}_0\mathbf{Y}_1$$

---

## 3. 인터랙티브 VQE 수소 분자 결합 에너지 시뮬레이터

(function() {
  var sR = document.getElementById('vqe-r-slider');
  var sTh = document.getElementById('vqe-th-slider');
  var rVal = document.getElementById('vqe-r-val');
  var thVal = document.getElementById('vqe-th-val');

  var eTxt = document.getElementById('vqe-e-txt');
  var e0Txt = document.getElementById('vqe-e0-txt');
  var errTxt = document.getElementById('vqe-err-txt');
  var svg = document.getElementById('vqe-svg');

  function morsePotential(r) {
    // Morse Potential for H2 ground state (in Hartree)
    var De = 0.174, a = 1.05, re = 0.7414, E_inf = -1.0;
    return De * Math.pow(1 - Math.exp(-a * (r - re)), 2) - De + E_inf;
  }

  function render() {
    var r = parseFloat(sR.value);
    var th = parseFloat(sTh.value);
    rVal.textContent = r.toFixed(2) + ' Å';
    thVal.textContent = 'θ = ' + th.toFixed(2) + ' rad';

    var e0 = morsePotential(r);
    // 안사츠 오차 추가: E(theta) = e0 + 0.1 * sin^2(theta - 0.2)
    var varErr = 0.15 * Math.pow(Math.sin(th - 0.2), 2);
    var eMeas = e0 + varErr;

    eTxt.textContent = eMeas.toFixed(3) + ' Ha';
    e0Txt.textContent = e0.toFixed(3) + ' Ha';
    errTxt.textContent = varErr.toFixed(4) + ' Ha';
    errTxt.style.color = varErr < 0.005 ? 'var(--chart-2)' : 'var(--destructive)';

    // SVG 렌더링
    svg.innerHTML = '';
    var ox = 40, oy = 120, wBox = 260, hBox = 90;

    // 모스 포텐셜 곡선 Path
    var dPath = '';
    for (var rv = 0.3; rv <= 2.5; rv += 0.05) {
      var ev = morsePotential(rv);
      var px = ox + ((rv - 0.3) / 2.2) * wBox;
      var py = oy - ((ev - (-1.2)) / 0.5) * hBox;
      dPath += (rv === 0.3 ? 'M ' : ' L ') + px + ' ' + py;
    }

    var path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    path.setAttribute('d', dPath);
    path.setAttribute('fill', 'none');
    path.setAttribute('stroke', 'var(--primary)');
    path.setAttribute('stroke-width', '2');
    svg.appendChild(path);

    // 현재 측정 점 Dot
    var curX = ox + ((r - 0.3) / 2.2) * wBox;
    var curY = oy - ((eMeas - (-1.2)) / 0.5) * hBox;

    var dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    dot.setAttribute('cx', curX); dot.setAttribute('cy', curY);
    dot.setAttribute('r', 5);
    dot.setAttribute('fill', 'var(--destructive)');
    dot.setAttribute('stroke', 'var(--background)');
    dot.setAttribute('stroke-width', '2');
    svg.appendChild(dot);
  }

  sR.addEventListener('input', render);
  sTh.addEventListener('input', render);
  render();
})();
