**SQD(Sample-based Quantum Diagonalization)**는 NISQ 양자 프로세서의 얕은 회로에서 추출한 **측정 표본 비트열(Bitstring Samples)**을 기저로 삼아, 고전 슈퍼컴퓨터에서 **축소된 유효 부분 공간(Effective Subspace)** 해밀토니안을 정확히 대각화하는 차세대 하이브리드 양자-고전 알고리즘이다.

## 2. VQE 대비 SQD의 핵심 차별점

1. **파라미터 최적화 루프 제거**: VQE처럼 수백 회의 하이브리드 최적화 반복(Gradient Descent)이 필요 없어 최적화 정체(Barren Plateaus) 문제 회피.
2. **다중 에너지 준위 계산**: 바닥 상태뿐만 아니라 1차, 2차 **여기 상태(Excited States)** 에너지를 단일 부분 공간 대각화로 일괄 도출.

## 3. 인터랙티브 SQD 부분 공간 크기 $K$에 따른 에너지 수렴 시뮬레이터

추출된 샘플 기저 개수 $K$ (10 ~ 200개)를 슬라이더로 조절하면 엄밀 대각화(Full Exact Diagonalization $2^{20} \approx 10^6$) 대비 바닥 상태 에너지 오차가 실시간으로 수렴한다.

<script>
(function() {
  var sK = document.getElementById('sqd-k-slider');
  var kVal = document.getElementById('sqd-k-val');
  var eTxt = document.getElementById('sqd-e-txt');
  var errTxt = document.getElementById('sqd-err-txt');
  var accTxt = document.getElementById('sqd-acc-txt');
  var svg = document.getElementById('sqd-svg');

  var eExact = -75.424; // Water molecule ground state approx

  function getEnergy(k) {
    // 오차는 k 증가에 따라 지수 감쇠: 0.15 * exp(-k / 25)
    var err = 0.15 * Math.exp(-k / 25.0);
    return eExact + err;
  }

  function render() {
    var k = parseInt(sK.value, 10);
    kVal.textContent = k + ' 개 기저';

    var eCalc = getEnergy(k);
    var err = Math.abs(eCalc - eExact);

    eTxt.textContent = eCalc.toFixed(4) + ' Ha';
    errTxt.textContent = err.toFixed(4) + ' Ha';

    // 1 kcal/mol = ~0.0016 Hartree (Chemical Accuracy threshold)
    if (err < 0.002) {
      accTxt.textContent = '✅ 화학적 정확도 달성 (< 1 kcal/mol)';
      accTxt.style.color = 'var(--chart-2)';
    } else {
      accTxt.textContent = '⚠️ 추가 기저 샘플링 필요';
      accTxt.style.color = 'var(--chart-1)';
    }

    // SVG 렌더링
    svg.innerHTML = '';
    var ox = 40, oy = 110, wBox = 260, hBox = 80;

    // 수렴 곡선 Path
    var dPath = '';
    for (var kv = 5; kv <= 150; kv += 5) {
      var ev = getEnergy(kv);
      var px = ox + ((kv - 5) / 145.0) * wBox;
      var py = oy - ((ev - eExact) / 0.15) * hBox;
      dPath += (kv === 5 ? 'M ' : ' L ') + px + ' ' + py;
    }

    var path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    path.setAttribute('d', dPath);
    path.setAttribute('fill', 'none');
    path.setAttribute('stroke', 'var(--primary)');
    path.setAttribute('stroke-width', '2');
    svg.appendChild(path);

    // 엄밀 바닥선 점선
    var baseLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
    baseLine.setAttribute('x1', ox); baseLine.setAttribute('y1', oy);
    baseLine.setAttribute('x2', ox + wBox); baseLine.setAttribute('y2', oy);
    baseLine.setAttribute('stroke', 'var(--chart-2)');
    baseLine.setAttribute('stroke-dasharray', '3 3');
    svg.appendChild(baseLine);

    // 현재 위치 Dot
    var curX = ox + ((k - 5) / 145.0) * wBox;
    var curY = oy - ((eCalc - eExact) / 0.15) * hBox;

    var dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    dot.setAttribute('cx', curX); dot.setAttribute('cy', curY);
    dot.setAttribute('r', 5);
    dot.setAttribute('fill', 'var(--destructive)');
    dot.setAttribute('stroke', 'var(--background)');
    dot.setAttribute('stroke-width', '2');
    svg.appendChild(dot);
  }

  sK.addEventListener('input', render);
  render();
})();
</script>
