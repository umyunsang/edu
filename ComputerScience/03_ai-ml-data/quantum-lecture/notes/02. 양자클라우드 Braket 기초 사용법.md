## 2. Amazon Braket Python SDK 벨 상태(Bell State) 구현

```python
from braket.circuits import Circuit
from braket.devices import LocalSimulator

# 1. 2-큐비트 벨 상태 회로 (|00> + |11>)/sqrt(2) 생성
bell_circ = Circuit().h(0).cnot(control=0, target=1)

print(bell_circ)
# T  : |0|1|
# q0 : -H-C-
# q1 : ---X-

# 2. 로컬 시뮬레이터 인스턴스화 및 1000 샷(Shots) 실행
device = LocalSimulator()
task = device.run(bell_circ, shots=1000)
result = task.result()

# 3. 측정 확률 카운트 출력
counts = result.measurement_counts
print("측정 결과:", counts) # {'00': ~500, '11': ~500}
```

(function() {
  var sShots = document.getElementById('brk-shots-slider');
  var shotsVal = document.getElementById('brk-shots-val');

  var c11Txt = document.getElementById('brk-c11-txt');

    var p11 = 1.0 - p00;

    var n00 = Math.round(p00 * shots);
    var n11 = shots - n00;

    c11Bar.style.width = (p11 * 100) + '%';

    errTxt.textContent = '± ' + (Math.abs(delta) * 100).toFixed(2) + ' %p';
  }

  sShots.addEventListener('input', render);
  render();
})();
</script>
