> [!abstract] 핵심 요약
> **공간으로 시간 벌기(Space-and-Time Tradeoffs)**는 추가적인 메모리 공간(테이블, 전처리 배열)을 사용하여 연산 시간(시간 복잡도)을 획기적으로 단축시키는 설계 전략이다.

---

## 1. 호스풀(Horspool) 문자열 매칭 이동표 시뮬레이터

패턴의 오른쪽 끝 문자부터 역방향으로 비교하며, 불일치 발생 시 패턴의 **착각 이동표(Shift Table)**를 참조하여 여러 칸을 한 번에 점프한다.

```html preview
  <div style="margin-bottom:14px">
    <label style="font-size:12px;font-weight:600;color:var(--muted-foreground);display:block;margin-bottom:4px">탐색할 패턴 (Pattern)</label>
    <input id="patInput" type="text" value="BARBER" style="width:100%;padding:8px;background:var(--background);color:var(--foreground);border:1px solid var(--border);border-radius:var(--radius);font-size:13px;font-family:monospace" />

  <div style="padding:12px;background:var(--background);border:1px solid var(--border);border-radius:var(--radius)">
    <div style="font-size:12px;font-weight:700;color:var(--chart-1);margin-bottom:8px">문자별 이동 거리 (Shift Distance):</div>

  <script>
    var pIn = document.getElementById('patInput');
    var stOut = document.getElementById('shiftTableOutput');

    function buildTable() {
      var pat = pIn.value.toUpperCase();
      var m = pat.length;
      if (m === 0) return;
      var table = {};

      for (var i = 0; i < m - 1; i++) {
        table[pat[i]] = m - 1 - i;
      }

      stOut.innerHTML = '';
      var chars = Object.keys(table);
      for (var k = 0; k < chars.length; k++) {
        var ch = chars[k];
        var card = document.createElement('div');
        card.style.cssText = 'padding:6px 12px;background:var(--card);border:1px solid var(--border);border-radius:var(--radius);text-align:center';
        card.innerHTML = '<div style="font-size:11px;color:var(--muted-foreground)">' + ch + '</div><div style="font-size:14px;font-weight:700;color:var(--primary)">' + table[ch] + '</div>';
        stOut.appendChild(card);
      }

      var defaultCard = document.createElement('div');
      defaultCard.style.cssText = 'padding:6px 12px;background:var(--card);border:1px solid var(--chart-2);border-radius:var(--radius);text-align:center';
      defaultCard.innerHTML = '<div style="font-size:11px;color:var(--muted-foreground)">기타 문자</div><div style="font-size:14px;font-weight:700;color:var(--chart-2)">' + m + '</div>';
      stOut.appendChild(defaultCard);
    }

    pIn.addEventListener('input', buildTable);
    buildTable();
  </script>
</div>
```
