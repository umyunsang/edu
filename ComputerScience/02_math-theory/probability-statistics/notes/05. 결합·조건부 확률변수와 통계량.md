다변량 데이터 분석에서는 두 개 이상의 확률변수가 결합된 \*\*결합확률분포(Joint Distribution)\*\*를 분석하여 변수 간의 상관관계와 조건부 기댓값을 도출한다.

---

## 1. 다변량 통계량과 분산 분해의 수학적 공식화

### 1) 공분산(Covariance)과 피어슨 상관계수(Correlation Coefficient)

$$\text{Cov}(X, Y) = \mathbb{E}\big[ (X - \mathbb{E}[X])(Y - \mathbb{E}[Y]) \big] = \mathbb{E}[X Y] - \mathbb{E}[X] \mathbb{E}[Y]$$

$$\rho_{X, Y} = \frac{\text{Cov}(X, Y)}{\sigma_X \sigma_Y} \in [-1, 1]$$

- 두 변수가 독립이면 $\text{Cov}(X, Y) = 0$ 및 $\rho\_{X,Y} = 0$ (역은 성립하지 않음).

### 2) 전조건부 기댓값 정리 (Adam's Law / Law of Total Expectation)

$$\mathbb{E}[Y] = \mathbb{E}_X\big[ \mathbb{E}_{Y|X}[Y \mid X] \big] = \sum_x \mathbb{E}[Y \mid X = x] \cdot P(X = x)$$

### 3) 전조건부 분산 분해 정리 (Eve's Law / Law of Total Variance)

$$\text{Var}(Y) = \mathbb{E}\big[ \text{Var}(Y \mid X) \big] + \text{Var}\big( \mathbb{E}[Y \mid X] \big)$$

(function() {
  var p1 = document.getElementById('cov-p1');
  var p2 = document.getElementById('cov-p2');
  var p3 = document.getElementById('cov-p3');
  var p4 = document.getElementById('cov-p4');
  var rCov = document.getElementById('cov-res-val');
  var rRho = document.getElementById('cov-res-rho');

  function parsePt(el) {
    var parts = el.value.split(',').map(function(s){ return parseFloat(s.trim()); });
    return [parts[0]||0, parts[1]||0];
  }

  function update() {
    var pts = [parsePt(p1), parsePt(p2), parsePt(p3), parsePt(p4)];
    var N = pts.length;
    var mx = pts.reduce(function(a,p){return a+p[0];}, 0) / N;
    var my = pts.reduce(function(a,p){return a+p[1];}, 0) / N;

    var cov = 0, vx = 0, vy = 0;
    pts.forEach(function(p) {
      cov += (p[0] - mx) * (p[1] - my);
      vx += (p[0] - mx) * (p[0] - mx);
      vy += (p[1] - my) * (p[1] - my);
    });
    cov /= N; vx /= N; vy /= N;
    var rho = (vx > 0 && vy > 0) ? (cov / Math.sqrt(vx * vy)) : 0;

    rCov.textContent = cov.toFixed(2);
    rRho.textContent = (rho >= 0 ? '+' : '') + rho.toFixed(2);
  }

  [p1, p2, p3, p4].forEach(function(el){ el.addEventListener('input', update); });
  update();
})();
</script>
