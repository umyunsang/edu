모집단의 정확한 모수(모평균 $\mu$, 모분산$\sigma^2$)를 알 수 없을 때, 관측된 표본 데이터를 바탕으로 통계적 가설을 검정하고 **부트스트랩(Bootstrap, 복원 재표본추출)**을 통해 추정량의 신뢰구간(Confidence Interval)을 도출한다.

> [!IMPORTANT]
>
> - **$p$-값 ($p$-value)**: 귀무가설$H_0$이 참이라는 가정 하에서, 실제로 관측된 통계량만큼(또는 그 이상으로) 극단적인 결과가 발생할 확률.
> - **부트스트랩(Bootstrap)**: 크기 $N$인 원래 표본에서 중복을 허용하여$N$개의 데이터를 복원 추출하는 과정을$B$번 반복하여 통계량의 경험적 표본분포를 형성.

(function() {
  var origSample = [12, 15, 18, 14, 22, 19, 13, 27];
  var bootMeans = [];

  var btnRun = document.getElementById('boot-btn-run');
  var btnReset = document.getElementById('boot-btn-reset');
  var counterEl = document.getElementById('boot-counter');
  var histEl = document.getElementById('boot-hist');
  var origMeanEl = document.getElementById('boot-orig-mean');
  var ciEl = document.getElementById('boot-ci-95');

  var sum = origSample.reduce(function(a, b) { return a + b; }, 0);
  var origMean = sum / origSample.length;
  origMeanEl.textContent = origMean.toFixed(2);

  function sampleWithReplacement(arr) {
    var res = [];
    for (var i = 0; i < arr.length; i++) {
      var idx = Math.floor(Math.random() * arr.length);
      res.push(arr[idx]);
    }
    return res;
  }

  function runBootstrap(times) {
    for (var t = 0; t < times; t++) {
      var sample = sampleWithReplacement(origSample);
      var m = sample.reduce(function(a, b) { return a + b; }, 0) / sample.length;
      bootMeans.push(m);
    }
    render();
  }

  function render() {
    counterEl.textContent = '누적 추출 횟수: ' + bootMeans.length + ' 회';
    if (bootMeans.length === 0) {
      histEl.innerHTML = '<div style="font-size:11px;color:var(--muted-foreground);margin:auto">버튼을 눌러 표본을 추출하세요.</div>';
      ciEl.textContent = '-';
      return;
    }

    bootMeans.sort(function(a, b) { return a - b; });
    var lowIdx = Math.floor(bootMeans.length * 0.025);
    var highIdx = Math.floor(bootMeans.length * 0.975);
    ciEl.textContent = '[ ' + bootMeans[lowIdx].toFixed(2) + ', ' + bootMeans[highIdx].toFixed(2) + ' ]';

    // 15개 빈 히스토그램
    var bins = new Array(15).fill(0);
    var minVal = 12, maxVal = 24;
    bootMeans.forEach(function(m) {
      var bIdx = Math.floor((m - minVal) / (maxVal - minVal) * 15);
      bIdx = Math.max(0, Math.min(14, bIdx));
      bins[bIdx]++;
    });

    var maxBin = Math.max.apply(null, bins);
    histEl.innerHTML = '';
    bins.forEach(function(cnt) {
      var col = document.createElement('div');
      col.style.flex = '1';
      col.style.height = '100%';
      col.style.display = 'flex';
      col.style.alignItems = 'flex-end';

      var bar = document.createElement('div');
      bar.style.width = '100%';
      bar.style.height = maxBin > 0 ? (cnt / maxBin * 90) + '%' : '0%';
      bar.style.background = 'var(--chart-1)';
      bar.style.borderRadius = '2px 2px 0 0';

      col.appendChild(bar);
      histEl.appendChild(col);
    });
  }

  btnRun.addEventListener('click', function() { runBootstrap(100); });
  btnReset.addEventListener('click', function() { bootMeans = []; render(); });

  runBootstrap(100);
})();

## 2. 가설검정 4단계 절차

1. **가설 수립**: 귀무가설 $H_0$vs 대립가설$H_1$.
2. **유의수준($\alpha$) 설정**: 통상$\alpha = 0.05$(5%) 또는$0.01$ (1%).
3. **검정통계량 및 $p$-값 도출**: 표본 데이터로부터 검정 통계량($Z, t, F$) 계산.
4. **의사결정**: $p\text{-value} \le \alpha$이면 귀무가설$H_0$ 기각(통계적으로 유의미한 차이 있음).
