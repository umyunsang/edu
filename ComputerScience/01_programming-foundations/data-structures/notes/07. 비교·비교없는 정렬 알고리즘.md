정렬(Sorting)은 원소들을 특정 기준(오름차순/내림차순)으로 재배열하는 알고리즘이다. 키 비교 기반 정렬은 이론적으로 $O(N \log N)$의 하한선을 가지며, 자릿수를 이용하는 **기수 정렬(Radix Sort)**은 $O(d \cdot N)$ 시간에 수행된다.

> [!IMPORTANT]
>
> - **안정 정렬(Stable Sort)**: 동일한 키 값을 가진 원소들의 상대적 순서가 정렬 후에도 유지됨 (삽입, 버블, 병합, 기수 정렬).
> - **불안정 정렬(Unstable Sort)**: 상대적 순서가 바뀔 수 있음 (선택, 퀵, 힙, 셸 정렬).
> - **제자리 정렬(In-Place Sort)**: 추가적인 배열 메모리를 $O(1)$만 사용 (선택, 삽입, 버블, 퀵 정렬).

(function() {
  var rawData = [45, 12, 89, 34, 67, 23, 78, 5];
  var arr = rawData.slice();
  var sortSelect = document.getElementById('sort-select');
  var btnStep = document.getElementById('btn-sort-step');
  var btnShuffle = document.getElementById('btn-sort-shuffle');
  var container = document.getElementById('sort-bar-container');
  var log = document.getElementById('sort-log');

  var i = 0, j = 0;
  var completed = false;

  function renderBars(comp1, comp2, sortedIdx) {
    container.innerHTML = '';
    var maxVal = Math.max.apply(null, arr);
    arr.forEach(function(val, idx) {
      var barWrap = document.createElement('div');
      barWrap.style.flex = '1';
      barWrap.style.maxWidth = '40px';
      barWrap.style.display = 'flex';
      barWrap.style.flexDirection = 'column';
      barWrap.style.alignItems = 'center';

      var valLabel = document.createElement('span');
      valLabel.style.fontSize = '11px';
      valLabel.style.fontWeight = '700';
      valLabel.style.marginBottom = '4px';
      valLabel.textContent = val;
      barWrap.appendChild(valLabel);

      var bar = document.createElement('div');
      bar.style.width = '100%';
      bar.style.height = (val / maxVal * 100) + '%';
      bar.style.borderRadius = '4px 4px 0 0';

      if (idx === comp1 || idx === comp2) {
        bar.style.background = 'var(--chart-5)'; // 비교/교환 중 (주황/빨강)
      } else if (sortedIdx !== undefined && idx >= sortedIdx) {
        bar.style.background = 'var(--chart-2)'; // 정렬 완료
      } else {
        bar.style.background = 'var(--chart-1)';
      }
      barWrap.appendChild(bar);
      container.appendChild(barWrap);
    });
  }

  function resetState() {
    i = 0;
    j = 0;
    completed = false;
    log.textContent = '배열 재설정 완료: [ ' + arr.join(', ') + ' ]';
    renderBars();
  }

  btnShuffle.addEventListener('click', function() {
    arr = [45, 12, 89, 34, 67, 23, 78, 5].sort(function() { return 0.5 - Math.random(); });
    resetState();
  });

  sortSelect.addEventListener('change', function() {
    resetState();
  });

  btnStep.addEventListener('click', function() {
    if (completed) {
      log.textContent = '정렬이 이미 완료되었습니다! 다시 시작하려면 섞기를 누르세요.';
      return;
    }

    var algo = sortSelect.value;
    var n = arr.length;

    if (algo === 'bubble') {
      if (i < n - 1) {
        if (j < n - 1 - i) {
          var swapped = false;
          if (arr[j] > arr[j + 1]) {
            var tmp = arr[j];
            arr[j] = arr[j + 1];
            arr[j + 1] = tmp;
            swapped = true;
          }
          log.textContent = '비교: arr[' + j + '](' + arr[j] + ') 와 arr[' + (j+1) + '](' + arr[j+1] + ')' + (swapped ? ' ➔ 교환됨' : ' ➔ 유지');
          renderBars(j, j + 1, n - i);
          j++;
        } else {
          j = 0;
          i++;
          if (i === n - 1) {
            completed = true;
            log.textContent = '버블 정렬 완료!';
            renderBars(-1, -1, 0);
          }
        }
      }
    } else if (algo === 'selection') {
      if (i < n - 1) {
        var minIdx = i;
        for (var k = i + 1; k < n; k++) {
          if (arr[k] < arr[minIdx]) minIdx = k;
        }
        var temp = arr[i];
        arr[i] = arr[minIdx];
        arr[minIdx] = temp;
        log.textContent = (i + 1) + '단계: 최솟값 arr[' + minIdx + '](' + arr[i] + ')을 arr[' + i + '] 위치로 교환';
        renderBars(i, minIdx, i);
        i++;
        if (i === n - 1) {
          completed = true;
          log.textContent = '선택 정렬 완료!';
          renderBars(-1, -1, 0);
        }
      }
    } else if (algo === 'insertion') {
      if (i < n) {
        if (i === 0) i = 1;
        var key = arr[i];
        var p = i - 1;
        while (p >= 0 && arr[p] > key) {
          arr[p + 1] = arr[p];
          p--;
        }
        arr[p + 1] = key;
        log.textContent = '삽입: key=' + key + ' 를 정렬된 앞부분 인덱스 ' + (p + 1) + '에 삽입';
        renderBars(p + 1, i, i);
        i++;
        if (i === n) {
          completed = true;
          log.textContent = '삽입 정렬 완료!';
          renderBars(-1, -1, 0);
        }
      }
    }
  });

  resetState();
})();
</script>

## 2. 정렬 알고리즘 복잡도 종합 비교

| 알고리즘 | 최선 시간 | 평균 시간 | 최악 시간 | 공간 복잡도 | 안정성 |
| :--- | :--- | :--- | :--- | :--- | :--- |
| **선택 정렬 (Selection)** | $O(N^2)$ | $O(N^2)$ | $O(N^2)$ | $O(1)$ | 불안정 |
| **삽입 정렬 (Insertion)** | $O(N)$ (거의 정렬 시) | $O(N^2)$ | $O(N^2)$ | $O(1)$ | **안정** |
| **버블 정렬 (Bubble)** | $O(N)$ | $O(N^2)$ | $O(N^2)$ | $O(1)$ | **안정** |
| **퀵 정렬 (Quick)** | $O(N \log N)$ | $O(N \log N)$ | $O(N^2)$ (편향 분할) | $O(\log N)$ | 불안정 |
| **병합 정렬 (Merge)** | $O(N \log N)$ | $O(N \log N)$ | $O(N \log N)$ | $O(N)$ | **안정** |
| **힙 정렬 (Heap)** | $O(N \log N)$ | $O(N \log N)$ | $O(N \log N)$ | $O(1)$ | 불안정 |
| **기수 정렬 (Radix)** | $O(d \cdot N)$ | $O(d \cdot N)$ | $O(d \cdot N)$ | $O(N + k)$ | **안정** |

## 3. 기수 정렬 (Radix Sort)의 특징

- 키 비교를 전혀 수행하지 않고, 각 자릿수(1의 자리 $\to$ 10의 자리 $\to$ 100의 자리, **LSD(Least Significant Digit)** 방식)별로 큐(Bucket)에 분배한 뒤 꺼내는 과정을 반복한다.
- 데이터의 자릿수 $d$가 작을 때 고속으로 동작하며, 큐의 FIFO 성질에 의해 **안정 정렬**을 만족한다.
