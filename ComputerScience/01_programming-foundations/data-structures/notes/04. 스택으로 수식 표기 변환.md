컴파일러와 인터프리터는 사람이 작성한 중위 표기식(`A + B * C`)을 연산자 우선순위와 괄호 처리가 단순한 **후위 표기식(`A B C * +`)**으로 변환하여 스택을 통해 단일 패스로 계산한다.

> [!IMPORTANT]
>
> - **피연산자(Operand)**: 만나는 즉시 출력 버퍼에 추가.
> - **연산자(Operator)**: 스택 Top의 연산자와 우선순위를 비교하여, 스택 Top 연산자의 우선순위가 자신보다 크거나 같으면 `pop()`하여 출력하고 자신을 `push()`.
> - **괄호 `(`, `)`**: 여는 괄호 `(`는 스택에 무조건 `push()`, 닫는 괄호 `)`를 만나면 여는 괄호 `(`가 나올 때까지 스택의 연산자를 모두 `pop()`하여 출력 후 `(`를 버린다.

<script>
(function() {
  var input = document.getElementById('expr-input');
  var btnStart = document.getElementById('btn-expr-start');
  var btnStep = document.getElementById('btn-expr-step');
  var btnReset = document.getElementById('btn-expr-reset');
  var currTokenEl = document.getElementById('curr-token');
  var opStackEl = document.getElementById('op-stack-view');
  var outPostfixEl = document.getElementById('output-postfix');
  var stepDescEl = document.getElementById('step-desc');

  var tokens = [];
  var stepIdx = 0;
  var opStack = [];
  var postfixOutput = [];

  function getPrecedence(op) {
    if (op === '+' || op === '-') return 1;
    if (op === '*' || op === '/') return 2;
    return 0; // for '('
  }

  function initSteps() {
    var raw = input.value.replace(/\s+/g, '');
    tokens = [];
    for (var i = 0; i < raw.length; i++) {
      var c = raw[i];
      if ('+-*/()'.indexOf(c) !== -1) {
        tokens.push(c);
      } else {
        var num = c;
        while (i + 1 < raw.length && '+-*/()'.indexOf(raw[i+1]) === -1) {
          num += raw[++i];
        }
        tokens.push(num);
      }
    }
    stepIdx = 0;
    opStack = [];
    postfixOutput = [];
    currTokenEl.textContent = '대기중';
    opStackEl.textContent = '[ ]';
    outPostfixEl.textContent = '(비어있음)';
    stepDescEl.textContent = '수식 토큰 분해 완료: [' + tokens.join(', ') + ']. Step 버튼을 누르세요.';
  }

  btnStart.addEventListener('click', initSteps);

  btnStep.addEventListener('click', function() {
    if (tokens.length === 0) initSteps();
    if (stepIdx >= tokens.length) {
      if (opStack.length > 0) {
        var popped = opStack.pop();
        postfixOutput.push(popped);
        currTokenEl.textContent = 'EOF (종료)';
        opStackEl.textContent = '[ ' + opStack.join(', ') + ' ]';
        outPostfixEl.textContent = postfixOutput.join(' ');
        stepDescEl.textContent = '입력 종료. 스택에 남은 연산자 [' + popped + ']를 출력으로 이동.';
      } else {
        stepDescEl.textContent = '변환 완료! 최종 후위 표기식: ' + postfixOutput.join(' ');
      }
      return;
    }

    var token = tokens[stepIdx++];
    currTokenEl.textContent = token;

    if ('+-*/'.indexOf(token) !== -1) {
      var prec = getPrecedence(token);
      var poppedList = [];
      while (opStack.length > 0 && getPrecedence(opStack[opStack.length - 1]) >= prec) {
        var p = opStack.pop();
        postfixOutput.push(p);
        poppedList.push(p);
      }
      opStack.push(token);
      stepDescEl.textContent = '연산자 [' + token + ']: 우선순위 비교 후 스택에 push. (방출된 연산자: ' + (poppedList.length ? poppedList.join(', ') : '없음') + ')';
    } else if (token === '(') {
      opStack.push('(');
      stepDescEl.textContent = '여는 괄호 \'(\': 스택에 무조건 push.';
    } else if (token === ')') {
      var popped = [];
      while (opStack.length > 0 && opStack[opStack.length - 1] !== '(') {
        var p = opStack.pop();
        postfixOutput.push(p);
        popped.push(p);
      }
      if (opStack.length > 0 && opStack[opStack.length - 1] === '(') {
        opStack.pop(); // discard '('
      }
      stepDescEl.textContent = '닫는 괄호 \')\': 여는 괄호까지 연산자 [' + popped.join(', ') + ']를 pop하여 출력에 추가 후 \'(\' 제거.';
    } else {
      postfixOutput.push(token);
      stepDescEl.textContent = '피연산자 (숫자/문자) [' + token + ']: 즉시 출력 버퍼에 추가.';
    }

    opStackEl.textContent = '[ ' + opStack.join(', ') + ' ]';
    outPostfixEl.textContent = postfixOutput.join(' ');
  });

  btnReset.addEventListener('click', function() {
    input.value = '( 3 + 5 ) * 2 - 4';
    initSteps();
  });

  initSteps();
})();
</script>

## 2. 연산자 우선순위(Precedence) 테이블

| 연산자 | ICP (In-Coming Precedence, 스택 진입 전) | ISP (In-Stack Precedence, 스택 내부) |
| :---: | :---: | :---: |
| `*`, `/` | 2 | 2 |
| `+`, `-` | 1 | 1 |
| `(` | 3 (가장 높음, 무조건 진입) | 0 (가장 낮음, 이후 연산자 진입 허용) |

## 3. 후위 표기법 계산 알고리즘 (Postfix Evaluation)

후위 표기식을 계산할 때는 피연산자용 스택을 사용한다:

1. 피연산자는 스택에 `push`한다.
2. 연산자를 만나면 스택에서 두 개의 피연산자 $B = \text{pop}()$,$A = \text{pop}()$를 꺼낸다.
3. $A \text{ op } B$연산을 수행한 후 결과를 다시 스택에 `push`한다. (순서 주의: 먼저 꺼낸 것이 오른쪽 피연산자$B$)
4. 수식 종료 후 스택에 남은 유일한 값이 최종 연산 결과다.
