C 언어는 시스템 제어와 하드웨어 직접 접근이 가능한 **절차 지향(Procedural) 고성능 프로그래밍 언어**다.

> [!IMPORTANT]
> - **연산자 우선순위**: 괄호 `()` $\to$ 단항 연산자 `!`, `~`, `++` $\to$ 산술 연산자 `*`, `/`, `%` $\to$ `+`, `-` $\to$ 시프트 `<<`, `>>` $\to$ 관계 비교 `<`, `>` $\to$ 등가 `==`, `!=` $\to$ 비트 AND `&` $\to$ XOR `^` $\to$ OR `|` $\to$ 논리 AND `&&` $\to$ 논리 OR `||` $\to$ 대입 `=`.
> - **단락 평가(Short-Circuit Evaluation)**: 논리식 `A && B`에서 $A$가 거짓이면 $B$는 평가하지 않고 즉시 거짓으로 판정.

```html preview
<script>
(function() {
  var inA = document.getElementById('bit-a');
  var inB = document.getElementById('bit-b');
  var selOp = document.getElementById('bit-op');
  var wrapB = document.getElementById('wrap-b');
  var rowB = document.getElementById('row-b');
  var dispA = document.getElementById('disp-bin-a');
  var dispB = document.getElementById('disp-bin-b');
  var dispRes = document.getElementById('disp-bin-res');
  var desc = document.getElementById('bit-desc');

  function toBin(n) {
    return (n.toString(2)).padStart(8, '0');
  }

  function calc() {
    var a = Math.max(0, Math.min(255, parseInt(inA.value, 10) || 0));
    var b = Math.max(0, Math.min(255, parseInt(inB.value, 10) || 0));
    var op = selOp.value;
    var res = 0;

    if (op === 'SHL' || op === 'SHR') {
      wrapB.style.display = 'none';
      rowB.style.display = 'none';
    } else {
      wrapB.style.display = 'flex';
      rowB.style.display = 'flex';
    }

    if (op === 'AND') {
      res = a & b;
      desc.textContent = '비트 AND (&): 두 비트가 모두 1일 때만 결과 비트가 1이 됩니다.';
    } else if (op === 'OR') {
      res = a | b;
      desc.textContent = '비트 OR (|): 두 비트 중 하나라도 1이면 결과 비트가 1이 됩니다.';
    } else if (op === 'XOR') {
      res = a ^ b;
      desc.textContent = '비트 XOR (^): 두 비트가 서로 다를 때만 1이 됩니다. (토글/암호화에 활용)';
    } else if (op === 'SHL') {
      res = (a << 1) & 0xFF;
      desc.textContent = 'Left Shift (<< 1): 모든 비트를 왼쪽으로 1칸 이동 (값 2배 곱셈 효과).';
    } else if (op === 'SHR') {
      res = (a >> 1) & 0xFF;
      desc.textContent = 'Right Shift (>> 1): 모든 비트를 오른쪽으로 1칸 이동 (값 2 나눗셈 몫 효과).';
    }

    dispA.textContent = toBin(a) + ' (' + a + ')';
    dispB.textContent = toBin(b) + ' (' + b + ')';
    dispRes.textContent = toBin(res) + ' (' + res + ')';
  }

  inA.addEventListener('input', calc);
  inB.addEventListener('input', calc);
  selOp.addEventListener('change', calc);

  calc();
})();
</script>
```

## 2. C 언어 연산자 종류 요약

| 분류 | 연산자 기호 | 설명 |  |  |
| :--- | :--- | :--- | - | - |
| **산술 연산자** | `+`, `-`, `*`, `/`, `%` | 덧셈, 뺄셈, 곱셈, 나눗셈(몫), 나머지 |  |  |
| **증감 연산자** | `++`, `--` | 전위(`++a`: 먼저 증가), 후위(`a++`: 참조 후 증가) |  |  |
| **관계 연산자** | `==`, `!=`, `<`, `>`, `<=`, `>=` | 조건 만족 시 1(참), 불만족 시 0(거짓) 반환 |  |  |
| **논리 연산자** | `&&` (AND), `||` (OR), `!` (NOT) | 논리 조건의 결합 및 부정 |
| **비트 연산자** | `&`, `|`, `^`, `\~`, `<<`, `>>` | 2진수 비트 단위의 하드웨어 레벨 조작 |  |
| **복합 대입** | `+=`, `-=`, `*=`, `/=`, `%=` | `a += 5` $\iff$ `a = a + 5` |  |  |
