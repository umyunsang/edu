파이썬 기본 프로그래밍 실습은 문자열 슬라이싱, 정수의 자릿수 분해, 조건 분기, 그리고 `while`/`for` 반복문을 활용한 핵심 알고리즘 문제 해결 능력을 기르는 과정이다.

> [!IMPORTANT]
>
> - **하샤드 수(Harshad Number)**: 양의 정수 $x$가 $x$의 각 자릿수의 합으로 나누어떨어지는 수 (예: $18 \to 1+8=9$, $18 \% 9 == 0$).
> - **콜라츠 추측(Collatz Conjecture)**: 짝수면 $n / 2$, 홀수면 $3n + 1$을 반복하여 1에 도달할 때까지의 연산 횟수를 추적 (500회 초과 시 -1 반환).

<script>
(function() {
  var mode = 'collatz';
  var tabCollatz = document.getElementById('py-tab-collatz');
  var tabHarshad = document.getElementById('py-tab-harshad');
  var inputN = document.getElementById('py-input-n');
  var btnRun = document.getElementById('py-btn-run');
  var resultOutput = document.getElementById('py-result-output');

  tabCollatz.addEventListener('click', function() {
    mode = 'collatz';
    tabCollatz.style.background = 'var(--primary)'; tabCollatz.style.color = '#fff';
    tabHarshad.style.background = 'transparent'; tabHarshad.style.color = 'var(--foreground)';
    inputN.value = 6;
    run();
  });

  tabHarshad.addEventListener('click', function() {
    mode = 'harshad';
    tabHarshad.style.background = 'var(--primary)'; tabHarshad.style.color = '#fff';
    tabCollatz.style.background = 'transparent'; tabCollatz.style.color = 'var(--foreground)';
    inputN.value = 18;
    run();
  });

  function run() {
    var n = parseInt(inputN.value, 10);
    if (isNaN(n) || n <= 0) {
      resultOutput.textContent = '1 이상의 양의 정수를 입력하세요.';
      return;
    }

    if (mode === 'collatz') {
      var seq = [n];
      var count = 0;
      var cur = n;
      while (cur !== 1 && count < 500) {
        if (cur % 2 === 0) {
          cur = Math.floor(cur / 2);
        } else {
          cur = cur * 3 + 1;
        }
        seq.push(cur);
        count++;
      }
      if (cur === 1) {
        resultOutput.innerHTML = seq.join(' ➔ ') + '<br><strong style="color:var(--chart-1)">✓ 도달 성공: 총 ' + count + ' 회 연산</strong>';
      } else {
        resultOutput.innerHTML = '<span style="color:var(--destructive)">500회 초과: 콜라츠 추측 실패 (-1 반환)</span>';
      }
    } else {
      var strN = n.toString();
      var sumDigits = 0;
      for (var i = 0; i < strN.length; i++) {
        sumDigits += parseInt(strN[i], 10);
      }
      var isHarshad = (n % sumDigits === 0);
      resultOutput.innerHTML = '입력값: ' + n + '<br>' +
                               '자릿수의 합: ' + strN.split('').join(' + ') + ' = ' + sumDigits + '<br>' +
                               n + ' % ' + sumDigits + ' = ' + (n % sumDigits) + '<br>' +
                               '<strong style="color:' + (isHarshad ? 'var(--chart-2)' : 'var(--destructive)') + '">' +
                               (isHarshad ? '✓ ' + n + '은(는) 하샤드 수입니다 (True)' : '✗ ' + n + '은(는) 하샤드 수가 아닙니다 (False)') + '</strong>';
    }
  }

  btnRun.addEventListener('click', run);
  run();
})();
</script>

## 2. 연습 문제 핵심 로직 패턴

| 문제 유형 | 핵심 파이썬 문법 | 예시 코드 |
| :--- | :--- | :--- |
| **문자 개수 세기** | `str.lower()`, `str.count()` | `s.lower().count('p') == s.lower().count('y')` |
| **자릿수 내림차순 정렬** | `sorted(..., reverse=True)`, `join()` | `int("".join(sorted(str(n), reverse=True)))` |
| **두 정수 사이의 합** | 등차수열의 합 공식 또는 `sum(range)` | `sum(range(min(a,b), max(a,b)+1))` |
| **구간 누적 곱/합** | 리스트 컴프리헨션 | `[x * i for i in range(1, n + 1)]` |

## 3. 파이썬 시간 절약 팁

- 문자열 대소문자 무시 비교 시 `s.lower()` 또는 `s.upper()`를 먼저 적용하여 순회한다.
- 숫자의 자릿수 분해는 `[int(c) for c in str(n)]` 리스트 컴프리헨션을 사용하면 간단하게 구현된다.
