</script>

</style>
</head>
<body>
<div id="sky">  <img id="followMouse" src="/static/photo3.png">  </div>
<script>
const i = document.getElementById("followMouse");
const d = document.getElementById("sky");
d.onmousemove = updateImage;
function updateImage(event) {
const X = event.clientX;
const Y = event.clientY;
i.style.transform = `translate(${X}px, ${Y}px)`;
}
</script>

const i = document.getElementById("largerImage");    
document.addEventListener('wheel', (e) => {     
if (e.deltaY < 0) {
i.width *= 1.1;        i.height *= 1.1;

<!DOCTYPE html>
<html><head>
<meta charset="UTF-8">  <title>예제9</title>  
</head>
<body>
<h4>좌우 화살표 키를 눌러 보세요</h4><hr>
<img id="man" src="/static/photo7.jpg">
<script>
const i = document.getElementById("man");
let x = 0;
document.addEventListener("keydown", (e) => {
if(e.key == "ArrowLeft") {
x -= 10;
if ( x < 0 ) x = 0;
} 
else if(e.key == "ArrowRight") {
x += 10;
if ( x > 450 ) x = 450;
}
i.style.transform = `translateX(${x}px)`;
});
</script>
</body></html>
